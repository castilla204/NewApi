## 🎯 TABLA CREADA EN POSTGRESQL

```sql
CREATE TABLE "UserMfaSettings" (
    "Id" integer NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    "UserId" integer NOT NULL,
    "IsEnabled" boolean NOT NULL DEFAULT false,
    "EncryptedSecret" character varying(256) NOT NULL,
    "Iv" character varying(64) NOT NULL,
    "EncryptedRecoveryCodes" character varying(1024),
    "RecoveryCodesIv" character varying(64),
    "EnabledAt" timestamp with time zone,
    "LastVerifiedAt" timestamp with time zone,
    "FailedAttempts" integer NOT NULL DEFAULT 0,
    "LockoutEnd" timestamp with time zone,
    "CreatedAt" timestamp with time zone NOT NULL DEFAULT NOW(),
    "UpdatedAt" timestamp with time zone NOT NULL DEFAULT NOW(),
    
    CONSTRAINT "PK_UserMfaSettings" PRIMARY KEY ("Id"),
    CONSTRAINT "FK_UserMfaSettings_Users_UserId" 
        FOREIGN KEY ("UserId") 
        REFERENCES "Users" ("Id") 
        ON DELETE CASCADE
);

CREATE UNIQUE INDEX "IX_UserMfaSettings_UserId" 
    ON "UserMfaSettings" ("UserId");

CREATE INDEX "IX_UserMfaSettings_IsEnabled" 
    ON "UserMfaSettings" ("IsEnabled");
```
